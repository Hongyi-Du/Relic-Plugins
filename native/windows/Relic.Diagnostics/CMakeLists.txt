cmake_minimum_required(VERSION 3.20)
project(RelicNativeStack LANGUAGES CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "Relic.Diagnostics requires Windows x64")
endif()
add_executable(Relic.NativeStack NativeStack.cpp)
target_compile_features(Relic.NativeStack PRIVATE cxx_std_20)
target_compile_definitions(Relic.NativeStack PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00)
target_compile_options(Relic.NativeStack PRIVATE /W4 /EHsc /utf-8)
target_link_libraries(Relic.NativeStack PRIVATE dbghelp user32)
